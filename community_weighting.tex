	\section*{Plots, coverages, and data frames}
	We will demonstrate one-level and two-level community weighting with a single plot. Table~\ref{tab:l1_plot} shows, a plot for the one-level example - there is a single subplot for which plot composition is reported. For the two-level example, Table~\ref{tab:l2_plot} shows a plot with \textit{two} subplots - in general there can be \textit{two or more} subplots per plot. The two-level refers to the fact that weighted traits are first computed at the subplot level and these subplot traits are weighted using subplot weights to produce a plot level trait value. One-level weighting is a special case of two-level weighting, where the there is one subplot. Table~\ref{tab:plots_coverages}, provides the employed notation. $A, B, C, D$ represent the cover classes (e.g. species, plant functional types). In Tables~\ref{tab:l1_plot},~\ref{tab:l2_plot}, $A_{r}^{s}$ represents the $r$-th replicate for cover class $A$ in the $s$-th subplot. In Table~\ref{tab:l2_plot}, there are $2$ replicates for cover class $A$ in subplot $1$ ($A_{1}^1, A_{2}^1$) and $2$ replicates for $D$ in subplot $2$ ($D_{1}^1, D_{2}^1$). The cover weight (\texttt{CoverWeight} column) for the various cover classes for the subplots are reported in Tables~\ref{tab:l1_cover},~\ref{tab:l2_cover}. $c_{k}^{s}$, represents the coverage weight for cover class $k$ in subplot $s$. Note that we include subplot identifier in the superscript!

	\section*{Community weighting and sampling}
	The following notation will be used:
	\begin{itemize}
		\item 
		${[\widetilde{c_k^s}]}_i$: $i$-th sample from a specified distribution $D_c(c_k^s)$. 
		\item 
		${[\widetilde{t_{k_r}^s}]}_i$: $i$-th sample from a specified distribution $D_t(\mu_{k_r}^s, \sigma_{k_r}^s)$.
	\end{itemize} 
	The community weighted trait for a plot can be obtained as follows:
	\begin{itemize} 
		\item For $i \in {1, \ldots, M}$:
		\begin{itemize} 
			\item For each subplot $s$:
			\begin{itemize}
				\item For every \texttt{CoverID}~$=k$:
				\begin{itemize} 
					\item Choose: $r$ (if replicates exist for \texttt{CoverID}).
					\item Sample: $({[\widetilde{c_k^s}]}_i, {[\widetilde{t_{k_r}^s}]}_i)$.
				\end{itemize}
				\item Compute: ${[t^s]}_i = \frac{\sum_k{{[\widetilde{c_k^s}]}_i} {[\widetilde{t_{k_r}^s}]}_i}{\sum_k{{[\widetilde{c_k^s}]}_i}}$ (subplot level weighting). 
			\end{itemize} 
		\item Compute $\widetilde{y_i} = \frac{\sum_s{w^s}{{[t^s]}_i}}{\sum_s{w^s}}$ (plot level weighting).
		\end{itemize}
		\item Compute:
		\begin{itemize} 
			\item $\widehat{\mu} = \mbox{\texttt{mean}}(\widetilde{y_1}, \ldots, \widetilde{y_M})$ (community weighted mean trait for plot), 
			\item $\widehat{\sigma} = \mbox{\texttt{sdev}}(\widetilde{y_1}, \ldots, \widetilde{y_M})$ (community weighted standard deviation of trait for plot).
		\end{itemize}
	\end{itemize}
	
	\textbf{Example (one-level case):} Let $A_{r=2}$ was chosen. Then,
	\begin{equation*}
		{[t^{1}]}_i = 
		\frac{ {[\widetilde{c_A^{1}}]}_i \colorbox{pink}{$ \textstyle {[\widetilde{t_{A_2}^{1}}]}_i  $} + 
			   {[\widetilde{c_B^{1}}]}_i {[\widetilde{t_{B_1}^{1}}]}_i + 
			   {[\widetilde{c_E^{1}}]}_i {[\widetilde{t_{E_1}^{1}}]}_i}
			 { {[\widetilde{c_A^{1}}]}_i + {[\widetilde{c_B^{1}}]}_i + {[\widetilde{c_E^{1}}]}_i}
	\end{equation*},
		
	\begin{equation*}
		\widetilde{y_i} = \frac{w^1{[t^{1}]}_i}{w^1}
	\end{equation*}
	
	\textbf{Example (two-level case):} Let $A_{r=1}$ and $D_{r=2}$ was chosen for subplots $s=1$ and $s=2$ respectively. Then,
		\begin{equation*}
			{[t^{1}]}_i = 
			\frac{ {[\widetilde{c_A^{1}}]}_i {[\widetilde{t_{A_1}^{1}}]}_i + 
				{[\widetilde{c_B^{1}}]}_i {[\widetilde{t_{B_1}^{1}}]}_i + 
				{[\widetilde{c_E^{1}}]}_i {[\widetilde{t_{E_1}^{1}}]}_i}
			{ {[\widetilde{c_A^{1}}]}_i + {[\widetilde{c_B^{1}}]}_i + {[\widetilde{c_E^{1}}]}_i}
		\end{equation*},
		
		\begin{equation*} 
		{[t^{2}]}_i = 
		\frac{ {[\widetilde{c_A^{2}}]}_i {[\widetilde{t_{A_1}^{2}}]}_i + 
			{[\widetilde{c_E^{2}}]}_i {[\widetilde{t_{E_1}^{2}}]}_i + 
			{[\widetilde{c_D^{2}}]}_i \colorbox{LightGreen}{$ \textstyle {[\widetilde{t_{D_2}^{2}}]}_i $} }
		{ {[\widetilde{c_A^{2}}]}_i + {[\widetilde{c_E^{2}}]}_i + {[\widetilde{c_D^{2}}]}_i}
		\end{equation*},
		 
		\begin{equation*} 
		\widetilde{y_i} = \frac{w^1{[t^{1}]}_i + w^2{[t^{2}]}_i}{w^1 + w^2}
		\end{equation*}

\section*{Distributions for sampling}
\subsection*{Trait sampling}
For generating the samples ${[\widetilde{t_{k_r}^s}]}_i$, the distribution $D_t$ is a truncated normal with mean and standard deviation: $(\mu_{k_r}^s, \sigma_{k_r}^s)$. Along with the two moments, the truncation support needs specification. Additionally, value clipping of the generated samples may be required (e.g. disallowing non-negative values). 

\subsection*{\texttt{CoverWeight} sampling}

\begin{itemize}
	\item \textbf{Percent cover}: If \texttt{CoverWeight} = $r$, $r \in [l_j, r_j]$, a \texttt{TrunNormal} distribution with mode $r$, standard deviation $d_j$ and support $[l_j, r_j]$ is used for sampling ${[\widetilde{c_k^s}]}_i$.

			$1$  & $l_1$ & $r_1$ & $d_1$ \\

	\item \textbf{Ordinal class}: If \texttt{CoverWeight} = $j$, a \texttt{Beta} distribution with mode $p_j$, concentration $q_j$, and support $[l_j, r_j]$ is used for sampling ${[\widetilde{c_k^s}]}_i$. The bin details are provided as in Table~\ref{tab:ordinal_cover}. 
		\begin{table}[htbp]

	\item \textbf{Role of bins}: For percent cover the bins are used for specifying a cover dependent uncertainty (via $d_j, l_j, r_j$) whereas in the case of ordinal class specficiation it defines the mode \textit{and} cover dependent uncertainty (mode is via $p_j$ and uncertainty via $q_j, j_j, r_j$).   
	\end{itemize}
